function updateHntChart() { var xhr = new XMLHttpRequest(); var url = "https://api.coingecko.com/api/v3/coins/helium/market_chart" + "?vs_currency=usd" + "&days=3" + "&interval=hourly"; xhr.open( "GET", url, true ); xhr.onreadystatechange = function() { if ( xhr.readyState != 4 ) { return; } if ( xhr.status != 200 ) { document.getElementById( "chartResult" ).innerHTML = "NO DATA"; return; } try { var result = JSON.parse( xhr.responseText ); if ( !result || !result.prices || result.prices.length < 2 ) { document.getElementById( "chartResult" ).innerHTML = "NO DATA"; return; } var points = []; var prices = result.prices; var i; /* Riduciamo i punti. Questo evita di sovraccaricare il vecchio browser Tizen. */ var step = Math.max( 1, Math.floor( prices.length / 60 ) ); for ( i = 0; i < prices.length; i += step ) { if ( prices[i] && prices[i].length >= 2 ) { points.push({ time: Number( prices[i][0] ), price: Number( prices[i][1] ) }); } } /* Assicuriamoci che l'ultimo prezzo sia presente. */ var last = prices[ prices.length - 1 ]; if ( last && last.length >= 2 ) { points.push({ time: Number( last[0] ), price: Number( last[1] ) }); } /* Controllo finale. */ if ( points.length >= 2 ) { drawChart( points ); } else { document.getElementById( "chartResult" ).innerHTML = "NO DATA"; } } catch (error) { document.getElementById( "chartResult" ).innerHTML = "ERROR"; console.log( "HNT chart:", error ); } }; xhr.onerror = function() { document.getElementById( "chartResult" ).innerHTML = "OFFLINE"; }; xhr.send(null); }
